## 一、项目概述

### 1.1 背景

spc-monitor 的预测模块当前存在 5 个核心诊断问题 + 6 个代码级缺陷，导致预测结果不可靠、风险评估严重漏判。经 15 项真实数据验证，确认了优化方向的正确性，并修正了原始方案中的 3 个关键假设。

### 1.2 升级目标

| 维度 | 当前状态 | 升级后 |
|------|---------|--------|
| 评估指标 | MAPE + R²（R² 全负失效） | MASE + 方向准确率（MAPE 降为参考） |
| 预测方法 | ETS/MA/ARIMA 手动选择 | ETS/ARIMA 自动选择（MA 淘汰） |
| 风险评估 | 点值比较（严重漏判） | 概率法越限概率 + CUSUM + Cpk 趋势 |
| API 完成度 | 30%（1/7 接口部分实现） | 100%（7/7 接口全部实现） |
| 数据持久化 | 无 | 3 张新表（预测结果/风险预测/换料分段） |

#### 关键代码逻辑

```python
def _naive_forecast(values: np.ndarray, horizon: int):
    """朴素基准：重复最后一个观测值"""
    last = float(values[-1])
    return [last] * horizon

def _calc_mase(actual: np.ndarray, predicted: np.ndarray) -> float:
    """MASE = MAE_model / MAE_naive；< 1 表示优于朴素基准"""
    naive_pred = np.full_like(actual, actual[0])  # 朴素预测：全部等于第一个值
    mae_naive = float(np.mean(np.abs(actual - naive_pred)))
    if mae_naive == 0:
        return float('inf')
    mae_model = float(np.mean(np.abs(actual - predicted)))
    return mae_model / mae_naive

def _calc_direction_accuracy(actual: np.ndarray, predicted: np.ndarray) -> float:
    """方向准确率：正确判断涨跌方向的比例"""
    if len(actual) < 2:
        return 0.0
    actual_dir = np.diff(actual) > 0
    pred_dir = np.diff(predicted) > 0
    return float(np.mean(actual_dir == pred_dir) * 100)
```

```python
from scipy.stats import norm

def _calc_breach_prob(predicted_mean: float, predicted_std: float,
                      usl: float | None, lsl: float | None) -> tuple[float, str]:
    """计算越限概率，同时考虑 USL 和 LSL，返回 (概率, 越限方向)"""
    prob_usl = 0.0
    prob_lsl = 0.0

    if usl is not None and predicted_std > 0:
        prob_usl = 1 - norm.cdf((usl - predicted_mean) / predicted_std)
    if lsl is not None and predicted_std > 0:
        prob_lsl = norm.cdf((lsl - predicted_mean) / predicted_std)

    if prob_usl >= prob_lsl:
        return prob_usl, 'upper'
    return prob_lsl, 'lower'

def _determine_risk_level(prob: float) -> str:
    """四级风险等级判定"""
    if prob < 0.05:   return 'LOW'
    if prob < 0.25:   return 'MEDIUM'
    if prob < 0.75:   return 'HIGH'
    return 'CRITICAL'

def _calc_breach_time(predictions, upper, lower, usl, lsl, times):
    """从预测序列中找到首次越限时间点"""
    for i, (u, lo) in enumerate(zip(upper, lower)):
        if usl is not None and u >= usl:
            return times[i] if i < len(times) else times[-1]
        if lsl is not None and lo <= lsl:
            return times[i] if i < len(times) else times[-1]
    return None  # 预测期内无越限
```

```python
from statsmodels.tsa.stattools import adfuller

def _adf_test(values: np.ndarray) -> tuple[float, bool]:
    """ADF 平稳性检验"""
    result = adfuller(values, maxlag=20, autolag='AIC')
    p_value = result[1]
    return p_value, p_value < 0.05

def _auto_select_d(values: np.ndarray, horizon: int) -> tuple[int, str]:
    """双拟合 d=0 和 d=1，取 MASE 更优者"""
    p_value, is_stationary = _adf_test(values)
    d_adf = 0 if is_stationary else 1

    split = int(len(values) * 0.8)
    train, actual = values[:split], values[split:]

    # 拟合 ARIMA(d=0)
    _, pred_d0 = _arima_forecast_with_d(train, len(actual), d=0)
    mase_d0 = _calc_mase(actual, np.array(pred_d0))

    # 拟合 ARIMA(d=1)
    _, pred_d1 = _arima_forecast_with_d(train, len(actual), d=1)
    mase_d1 = _calc_mase(actual, np.array(pred_d1))

    # 差距 < 5% 时选 ADF 推荐的 d（更简洁）
    if abs(mase_d0 - mase_d1) / max(mase_d0, mase_d1) < 0.05:
        return d_adf, f"ADF p={p_value:.4f}, d={d_adf}（差距<5%选简洁模型）"

    if mase_d0 <= mase_d1:
        return 0, f"ADF p={p_value:.4f}, d=0 MASE={mase_d0:.3f} 最优"
    return 1, f"ADF p={p_value:.4f}, d=1 MASE={mase_d1:.3f} 最优"

def _auto_select_model(values: np.ndarray, horizon: int) -> tuple[str, str]:
    """自动选择最优模型：ETS vs ARIMA（双 d）"""
    split = int(len(values) * 0.8)
    train, actual = values[:split], values[split:]
    results = {}

    # ETS
    pred_ets, _, _ = _ets_forecast(train, len(actual))
    results['ets'] = _calc_mase(actual, np.array(pred_ets[:len(actual)]))

    # ARIMA(d=0)
    pred_d0, _, _ = _arima_forecast_with_d(train, len(actual), d=0)
    results['arima_d0'] = _calc_mase(actual, np.array(pred_d0[:len(actual)]))

    # ARIMA(d=1)
    pred_d1, _, _ = _arima_forecast_with_d(train, len(actual), d=1)
    results['arima_d1'] = _calc_mase(actual, np.array(pred_d1[:len(actual)]))

    best = min(results, key=results.get)
    reason = f"MASE={results[best]:.3f}（最优）"
    return best, reason
```

#### 关键代码逻辑：滑动窗口 Cpk

```python
def _sliding_window_cpk(values: np.ndarray, usl: float, lsl: float,
                        window: int = 100) -> dict:
    """滑动窗口 Cpk 计算"""
    cpk_series = []
    for i in range(window, len(values) + 1, window):
        segment = values[i - window:i]
        mu = np.mean(segment)
        sigma = np.std(segment, ddof=1)
        if sigma == 0:
            cpk_series.append(float('inf'))
        else:
            cpk = min((usl - mu) / (3 * sigma), (mu - lsl) / (3 * sigma))
            cpk_series.append(round(cpk, 3))

    # 趋势判定：最后 3 个窗口 vs 最前 3 个窗口
    if len(cpk_series) >= 6:
        early = np.mean(cpk_series[:3])
        late = np.mean(cpk_series[-3:])
        trend = 'rising' if late > early * 1.05 else ('falling' if late < early * 0.95 else 'stable')
    else:
        trend = 'stable'

    # 能力评价
    current = cpk_series[-1] if cpk_series else 0
    if current < 0.5:    capability = 'severe_insufficient'
    elif current < 1.0:  capability = 'insufficient'
    elif current < 1.33: capability = 'sufficient'
    else:                capability = 'excellent'

    return {
        'cpk_current': current,
        'cpk_trend': trend,
        'cpk_window': cpk_series,
        'window_size': window,
        'capability': capability,
    }
```

```python
def _parse_segments(data: list[dict]) -> list[dict]:
    """解析换料/换罐标记，返回分段边界"""
    segments = []
    current_start = 0
    for i, row in enumerate(data):
        marker = row.get('remark', '') or row.get('sample_id', '')
        if 'HG' in marker or '换罐' in marker:
            if i > current_start:
                segments.append({
                    'start': current_start,
                    'end': i,
                    'marker': marker,
                })
            current_start = i
    # 最后一段
    segments.append({
        'start': current_start,
        'end': len(data),
        'marker': None,
    })
    return segments

def _segmented_cusum(values: np.ndarray, segments: list[dict],
                     threshold: float = 5.0, drift: float = 0.5) -> list[dict]:
    """每段独立计算标准 CUSUM"""
    results = []
    for seg in segments:
        seg_values = values[seg['start']:seg['end']]
        if len(seg_values) < 10:
            continue
        mu = np.mean(seg_values)
        sigma = np.std(seg_values, ddof=1)
        if sigma == 0:
            continue

        s_pos = s_neg = 0.0
        max_drift = 0.0
        for v in seg_values:
            z = (v - mu) / sigma
            s_pos = max(0, s_pos + z - drift)
            s_neg = min(0, s_neg + z + drift)
            max_drift = max(max_drift, abs(s_pos), abs(s_neg))

        direction = 'up' if s_pos > abs(s_neg) else ('down' if abs(s_neg) > s_pos else 'stable')
        results.append({
            'start': seg['start'],
            'end': seg['end'],
            'marker': seg['marker'],
            'cusum_drift': round(max_drift, 3),
            'direction': direction,
            'alert': max_drift > threshold,
        })
    return results
```

## 六、关键技术决策

| 决策 | 结论 | 理由 |
|------|------|------|
| 预测方法 | 仅保留 ETS + ARIMA | MA 的 MASE 为 5-12，无增量价值 |
| ARIMA d 参数 | 双拟合 d=0 和 d=1，取 MASE 更优 | ADF 在边界情况下不够稳健 |
| 越限概率公式 | `max(prob_USL, prob_LSL)` | 同时考虑上下规格限方向 |
| Cpk 趋势 | 滑动窗口 100 点 | 比前段/后段二分法更精细 |
| CI 计算 | 统一为 `residual_std * sqrt(step)` | 三种方法可比 |
| 缓存策略 | 5 分钟，按 `(indicator, product, horizon, model)` 缓存 | 与"每次重新拟合"不矛盾：缓存内复用，超时重拟 |
| 模型训练 | 每次重新拟合 | 最简单，不需模型持久化 |
| 关联假设 | 正相关（修正） | 实测脂肪↔蛋白质 r=0.78 |
| 低 CV 数据 | CV < 2% 时回退朴素基准 | 4% 的数据集模型无增量价值 |
| 依赖引入 | P3 引入 sklearn | 前两阶段用 numpy/scipy/statsmodels 足够 |

## 七、风险与注意事项

| 风险 | 影响 | 缓解措施 |
|------|------|---------|
| FT1 数据量增长后 SQLite 性能 | P2 持久化写入增加 | 加索引，单表数据量预计 < 10 万行，SQLite 够用 |
| sklearn 增加打包体积 30-50MB | Tauri 安装包变大 | P3 引入，前期不引入 |
| 换料标记格式不统一 | CUSUM 分段可能遗漏 | 先解析 HG/换罐，逐步补充其他标记 |
| 酸度模型 MASE > 1 | 模型无增量价值 | P1 验证不同步数下是否有局部价值，否则回退朴素基准 |
| 概率法计算需要 spec limits | 部分数据集无规格限 | 仅对有规格限的指标计算越限概率 |
